##################################################
from fastapi import FastAPI, Form
from fastapi.middleware.cors import CORSMiddleware
from fastapi.responses import FileResponse
import csv
######################################################

ستورد مكتبة FastAPI لإنشاء السيرفر
Form لاستقبال بيانات الاستمارة،
مكتبة csv للتعامل مع ملف البيانات
FileResponse لإرجاع هذه الصفحة

######################################################
app = FastAPI()
#####################################################
نقوم بإنشاء التطبيق

#####################################################
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)
######################################################
تفعيل إعدادات السماح 
(CORS) لتمكين المتصفح من إرسال الطلبات للسيرفر من أي مصدر دون مشاكل أمان.


#######################################################3
@app.post("/register")
######################################################
نحدد المسار /register

#################################################
async def register(
    full_name: str = Form(...),
    phone: int = Form(...),
    email: str = Form(...),
    password: int = Form(...),
    city: str = Form(...)
):
###########################################
باستقبال طلبات POST لاستقبال بيانات المدخلات من استمارة 
HTML بنجاح

######################################################
headers = ["Name", "Phone", "Email", "Password", "City"]
user_data = [full_name, phone, email, password, city]

with open("users.csv", mode="a", newline="", encoding="utf-8-sig") as file:
    writer = csv.writer(file)

    if file.tell() == 0:
        writer.writerow(headers)

    writer.writerow(user_data)
############################################################
نفتح الملف users.csv للإضافة،
 ونتحقق إن كان الملف فارغاً لنكتب العناوين (Headers)
  أولاً ثم نضيف صف البيانات الجديد
######################################################
print("The data has been successfully saved to the users.csv file!")

return FileResponse("done.html")
